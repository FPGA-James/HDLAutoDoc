ENTITY traffic_light IS¶

Traffic light controller.

Implements a standard UK traffic light sequencer with four states: RED, RED_AMBER, GREEN, and AMBER. The FSM advances on each rising clock edge when timer_exp is asserted.

State sequence: RED -> RED_AMBER -> GREEN -> AMBER -> RED.

src/traffic_light.vhd¶
  1library ieee;
  2use ieee.std_logic_1164.all;
  3use ieee.numeric_std.all;
  4
  5-- Traffic light controller.
  6--
  7-- Implements a standard UK traffic light sequencer with four states:
  8-- RED, RED_AMBER, GREEN, and AMBER. The FSM advances on each rising
  9-- clock edge when timer_exp is asserted.
 10--
 11-- State sequence: RED -> RED_AMBER -> GREEN -> AMBER -> RED.
 12entity traffic_light is
 13    generic (
 14        -- System clock frequency in Hz. Used to derive phase timer counts.
 15        CLK_FREQ_HZ  : natural := 50_000_000;
 16        -- Duration of the RED and GREEN phases in milliseconds.
 17        PHASE_MS     : natural := 5_000;
 18        -- Duration of the RED_AMBER and AMBER transition phases in milliseconds.
 19        TRANS_MS     : natural := 2_000
 20    );
 21    port (
 22        -- System clock, rising-edge triggered.
 23        clk       : in  std_logic;
 24        -- Synchronous active-high reset. Returns FSM to RED state.
 25        rst       : in  std_logic;
 26        -- Phase timer expiry pulse. Must be high for exactly one cycle.
 27        timer_exp : in  std_logic;
 28        -- Async active-low power-on reset for the phase watchdog counter.
 29        -- Separate from rst so the counter clears at power-on regardless of
 30        -- when the system reset deasserts.
 31        por_n     : in  std_logic;
 32        -- Red lamp drive. Asserted in RED and RED_AMBER states.
 33        red_out   : out std_logic;
 34        -- Amber lamp drive. Asserted in RED_AMBER and AMBER states.
 35        amber_out : out std_logic;
 36        -- Green lamp drive. Asserted in GREEN state only.
 37        green_out : out std_logic
 38    );
 39end entity traffic_light;
 40
 41-- RTL implementation using a 3-process Moore FSM style.
 42architecture rtl of traffic_light is
 43
 44    -- FSM state type encoding all four traffic light phases.
 45    type t_state is (
 46        -- Red light only. Vehicles must remain stopped.
 47        RED,
 48        -- Red and amber simultaneously. Vehicles prepare to move.
 49        RED_AMBER,
 50        -- Green light. Vehicles may proceed.
 51        GREEN,
 52        -- Amber light only. Vehicles prepare to stop.
 53        AMBER
 54    );
 55
 56    -- Current state register, initialised to RED at power-on.
 57    signal state      : t_state := RED;
 58    -- Next state combinational signal, feeds state register.
 59    signal next_state : t_state;
 60
 61    -- Phase watchdog counter. Counts clock cycles spent in the current state.
 62    -- Cleared by por_n (async) or whenever the FSM transitions to RED.
 63    signal phase_cnt  : std_logic_vector(15 downto 0);
 64
 65begin
 66
 67    -- p_state_reg: Clocked state register.
 68    --
 69    -- Captures next_state on each rising edge of clk.
 70    -- When rst is asserted the state returns to RED synchronously,
 71    -- regardless of timer_exp or next_state.
 72    --
 73    -- .. wavedrom::
 74    --
 75    --    { "signal": [
 76    --      { "name": "clk",   "wave": "P......." },
 77    --      { "name": "rst",   "wave": "1.0....." },
 78    --      { "name": "state", "wave": "=.=.....", "data": ["(any)","RED"] },
 79    --      { "name": "red",   "wave": "x.1....." }
 80    --    ]}
 81    p_state_reg : process(clk)
 82    begin
 83        if rising_edge(clk) then
 84            if rst = '1' then
 85                state <= RED;
 86            else
 87                state <= next_state;
 88            end if;
 89        end if;
 90    end process p_state_reg;
 91
 92    -- p_next_state: Combinational next-state logic.
 93    --
 94    -- Evaluates the FSM transition function. Default assignment
 95    -- next_state <= state implements hold when timer_exp is low.
 96    -- All four transitions are guarded solely by timer_exp = 1.
 97    --
 98    -- Transitions:
 99    --   RED       + timer_exp -> RED_AMBER
100    --   RED_AMBER + timer_exp -> GREEN
101    --   GREEN     + timer_exp -> AMBER
102    --   AMBER     + timer_exp -> RED
103    --
104    -- .. wavedrom::
105    --
106    --    { "signal": [
107    --      { "name": "clk",       "wave": "P........." },
108    --      { "name": "timer_exp", "wave": "0.1.1.1.1." },
109    --      { "name": "state",     "wave": "=.=.=.=.=.",
110    --        "data": ["RED","RED_AMB","GREEN","AMBER","RED"] },
111    --      { "name": "red_out",   "wave": "1.1.0.0.1." },
112    --      { "name": "amber_out", "wave": "0.1.0.1.0." },
113    --      { "name": "green_out", "wave": "0.0.1.0.0." }
114    --    ]}
115    p_next_state : process(state, timer_exp)
116    begin
117        next_state <= state;
118
119        case state is
120            when RED =>
121                if timer_exp = '1' then
122                    next_state <= RED_AMBER;
123                end if;
124
125            when RED_AMBER =>
126                if timer_exp = '1' then
127                    next_state <= GREEN;
128                end if;
129
130            when GREEN =>
131                if timer_exp = '1' then
132                    next_state <= AMBER;
133                end if;
134
135            when AMBER =>
136                if timer_exp = '1' then
137                    next_state <= RED;
138                end if;
139        end case;
140    end process p_next_state;
141
142    -- p_outputs: Combinational Moore output decode.
143    --
144    -- Drives lamp outputs based solely on current state.
145    -- All outputs default to 0 before the case statement to
146    -- prevent latches during synthesis.
147    --
148    -- Output truth table:
149    --   RED       -> red=1 amber=0 green=0
150    --   RED_AMBER -> red=1 amber=1 green=0
151    --   GREEN     -> red=0 amber=0 green=1
152    --   AMBER     -> red=0 amber=1 green=0
153    --
154    -- .. wavedrom::
155    --
156    --    { "signal": [
157    --      { "name": "state",     "wave": "=.=.=.=.",
158    --        "data": ["RED","RED_AMB","GREEN","AMBER"] },
159    --      { "name": "red_out",   "wave": "1.1.0.0." },
160    --      { "name": "amber_out", "wave": "0.1.0.1." },
161    --      { "name": "green_out", "wave": "0.0.1.0." }
162    --    ]}
163    p_outputs : process(state)
164    begin
165        red_out   <= '0';
166        amber_out <= '0';
167        green_out <= '0';
168
169        case state is
170            -- define red
171            when RED       => red_out <= '1';
172            when RED_AMBER => red_out <= '1'; amber_out <= '1';
173            when GREEN     => green_out <= '1';
174            when AMBER     => amber_out <= '1';
175        end case;
176    end process p_outputs;
177
178    -- p_watchdog: Phase watchdog counter.
179    --
180    -- Increments each clock cycle to track how long the FSM has been in the
181    -- current phase. Resets to zero whenever the FSM returns to RED.
182    --
183    -- Uses por_n (async active-low) rather than rst so the counter is
184    -- guaranteed zero at power-on before rst has been asserted by the system.
185    -- This creates a deliberate reset domain crossing: state is driven under
186    -- rst (synchronous) but read here under por_n (asynchronous).
187    p_watchdog : process(clk, por_n) is
188    begin
189        if por_n = '0' then
190            phase_cnt <= (others => '0');
191        elsif rising_edge(clk) then
192            if state = RED then
193                phase_cnt <= (others => '0');
194            else
195                phase_cnt <= std_logic_vector(unsigned(phase_cnt) + 1);
196            end if;
197        end if;
198    end process p_watchdog;
199
200end architecture rtl;
