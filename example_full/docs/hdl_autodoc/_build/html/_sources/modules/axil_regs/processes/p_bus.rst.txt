       always_ff @(posedge clk) begin : p_bus
           if (rst) begin
               ctrl_r      <= 32'd0;
               threshold_r <= 32'd0;
               scratch_r   <= 32'd0;
               rdata_r     <= 32'd0;
               bvalid_r    <= 1'b0;
               rvalid_r    <= 1'b0;
           end else begin
               if (wr_fire) begin
                   bvalid_r <= 1'b1;
                   case (awaddr[4:2])
                       3'd0:    ctrl_r      <= wdata;
                       3'd2:    threshold_r <= wdata;
                       3'd3:    scratch_r   <= wdata;
                       default: ;
                   endcase
               end else if (s_axil.bready) begin
                   bvalid_r <= 1'b0;
               end
   
               if (rd_fire) begin
                   rvalid_r <= 1'b1;
                   case (araddr[4:2])
                       3'd0:    rdata_r <= ctrl_r;
                       3'd1:    rdata_r <= {29'd0, overflow, state_code};
                       3'd2:    rdata_r <= threshold_r;
                       3'd3:    rdata_r <= scratch_r;
                       default: rdata_r <= {16'd0, count};
                   endcase
               end else if (s_axil.rready) begin
                   rvalid_r <= 1'b0;
               end
           end
       end
