.. code-block:: vhdl

       p_phase : process(clk) is
       begin
           if rising_edge(clk) then
               if rst = '1' then
                   phase <= IDLE;
               else
                   case phase is
                       when IDLE =>
                           if s_apb_psel = '1' and s_apb_penable = '0' then
                               addr  <= s_apb_paddr;
                               wdata <= s_apb_pwdata;
                               if s_apb_pwrite = '1' then
                                   phase <= WR_ADDR;
                               else
                                   phase <= RD_ADDR;
                               end if;
                           end if;
                       when WR_ADDR =>
                           if m_axil_awready = '1' then
                               phase <= WR_RESP;
                           end if;
                       when WR_RESP =>
                           if m_axil_bvalid = '1' then
                               phase <= FINISH;
                           end if;
                       when RD_ADDR =>
                           if m_axil_arready = '1' then
                               phase <= RD_DATA;
                           end if;
                       when RD_DATA =>
                           if m_axil_rvalid = '1' then
                               rdata <= m_axil_rdata;
                               phase <= FINISH;
                           end if;
                       when FINISH =>
                           if s_apb_penable = '1' then
                               phase <= IDLE;
                           end if;
                   end case;
               end if;
           end if;
       end process p_phase;
