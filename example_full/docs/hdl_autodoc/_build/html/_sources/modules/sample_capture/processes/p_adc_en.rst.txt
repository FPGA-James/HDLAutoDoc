.. code-block:: systemverilog

       always_ff @(posedge adc_clk) begin : p_adc_en
           if (adc_rst) begin
               en_meta  <= 1'b0;
               en_sync  <= 1'b0;
               clr_meta <= 1'b0;
               clr_sync <= 1'b0;
