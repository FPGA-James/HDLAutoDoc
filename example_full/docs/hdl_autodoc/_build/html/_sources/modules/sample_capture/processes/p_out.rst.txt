.. code-block:: systemverilog

       always_ff @(posedge clk or negedge rst_n) begin : p_out
           if (!rst_n) begin
               sample_valid <= 1'b0;
               sample_data  <= '0;
           end else if (rd_en) begin
               sample_valid <= 1'b1;
               sample_data  <= fifo_rd_data;
           end else if (sample_ready) begin
               sample_valid <= 1'b0;
           end
       end
