       always_ff @(posedge clk or negedge rst_n) begin : p_ctrl
           if (!rst_n) begin
               capture_en <= 1'b0;
               clear_req  <= 1'b0;
           end else begin
               capture_en <= enable;
               clear_req  <= clear_overflow;
           end
