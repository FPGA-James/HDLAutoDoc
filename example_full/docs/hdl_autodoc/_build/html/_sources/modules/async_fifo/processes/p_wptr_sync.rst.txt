.. code-block:: vhdl

       p_wptr_sync : process(rd_clk, rd_rst_n) is
       begin
           if rd_rst_n = '0' then
               wptr_meta  <= (others => '0');
               wptr_rsync <= (others => '0');
           elsif rising_edge(rd_clk) then
               wptr_meta  <= wptr_gray;
               wptr_rsync <= wptr_meta;
           end if;
       end process p_wptr_sync;
