src/top.vhd¶
  1library ieee;
  2use ieee.std_logic_1164.all;
  3
  4-- Sampler top level.
  5--
  6-- Captures samples from a converter clock into the system clock, runs a capture
  7-- window under register control (APB in, through an AXI4-Lite register file)
  8-- and sends the samples, and a second external byte stream, through packet
  9-- filters. The converter and system clocks are asynchronous to each other.
 10--
 11-- .. mermaid::
 12--    :caption: Top-level structure
 13--
 14--    flowchart LR
 15--      APB[s_apb] --> BR[apb_bridge] --> RG[axil_adapter + axil_regs]
 16--      RG -- start --> CT[sample_ctrl]
 17--      ADC[adc_data] --> CAP[sample_capture]
 18--      CT -- capture_en --> CAP
 19--      CAP --> SS[sample_stream] --> MA[m_axis]
 20--      EXT[s_ext] --> AP[axis_pipeline] --> ME[m_ext]
 21--      RG -- filter_enable, threshold --> SS
 22--      RG -- filter_enable, threshold --> AP
 23entity top is
 24    generic (
 25        -- Length of the capture window in system clock cycles.
 26        WINDOW_CYCLES : positive := 1024
 27    );
 28    port (
 29        -- System clock.
 30        clk            : in  std_logic;
 31        -- System-domain asynchronous active-low reset input.
 32        arst_n         : in  std_logic;
 33        -- Converter clock; asynchronous to clk.
 34        adc_clk        : in  std_logic;
 35        -- Converter-domain asynchronous active-low reset input.
 36        adc_arst_n     : in  std_logic;
 37        -- Sample from the converter.
 38        adc_data       : in  std_logic_vector(11 downto 0);
 39        -- APB subordinate port for register access.
 40        s_apb_psel     : in  std_logic;
 41        s_apb_penable  : in  std_logic;
 42        s_apb_pwrite   : in  std_logic;
 43        s_apb_paddr    : in  std_logic_vector(7 downto 0);
 44        s_apb_pwdata   : in  std_logic_vector(31 downto 0);
 45        s_apb_prdata   : out std_logic_vector(31 downto 0);
 46        s_apb_pready   : out std_logic;
 47        -- Filtered captured samples (low byte, one-byte packets).
 48        m_axis_tdata   : out std_logic_vector(7 downto 0);
 49        m_axis_tvalid  : out std_logic;
 50        m_axis_tready  : in  std_logic;
 51        m_axis_tlast   : out std_logic;
 52        -- External byte stream in.
 53        s_ext_tdata    : in  std_logic_vector(7 downto 0);
 54        s_ext_tvalid   : in  std_logic;
 55        s_ext_tready   : out std_logic;
 56        s_ext_tlast    : in  std_logic;
 57        -- External byte stream out, after filtering.
 58        m_ext_tdata    : out std_logic_vector(7 downto 0);
 59        m_ext_tvalid   : out std_logic;
 60        m_ext_tready   : in  std_logic;
 61        m_ext_tlast    : out std_logic;
 62        -- High while a capture is in progress.
 63        led_busy       : out std_logic;
 64        -- High after a sample was lost to a full FIFO.
 65        led_overflow   : out std_logic
 66    );
 67end entity top;

184    -- System-domain reset.
185    u_rst_sys : entity work.reset_sync
186        port map (clk => clk, arst_n => arst_n, srst_n => srst_n);
187
188    -- Converter-domain reset.
189    u_rst_adc : entity work.reset_sync
190        port map (clk => adc_clk, arst_n => adc_arst_n, srst_n => adc_srst_n);
191
192    rst     <= not srst_n;
193    adc_rst <= not adc_srst_n;

195    -- p_ovf_sync: Brings the converter-domain overflow flag onto clk.
196    p_ovf_sync : process(clk) is
197    begin
198        if rising_edge(clk) then
199            ovf_meta <= overflow;
200            ovf_sync <= ovf_meta;
201        end if;
202    end process p_ovf_sync;
203
204    led_busy     <= busy;
205    led_overflow <= ovf_sync;
